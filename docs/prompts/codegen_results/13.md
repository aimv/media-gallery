```dockerfile
# Dockerfile.api
# Stage 1: builder
FROM golang:1.26-alpine AS builder
WORKDIR /app
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -o /api ./cmd/api

# Stage 2: runner
FROM alpine:latest
RUN apk --no-cache add ca-certificates
WORKDIR /app
COPY --from=builder /api .
COPY --from=builder /app/internal/infrastructure/persistence/postgres/migrations ./internal/infrastructure/persistence/postgres/migrations
EXPOSE 8080
CMD ["./api"]
```

    environment:
      POSTGRES_DB: media_gallery
      POSTGRES_USER: media_user
      POSTGRES_PASSWORD: media_pass
    ports:
      - "5432:5432"
    volumes:

    healthcheck:

      interval: 5s

      retries: 5
