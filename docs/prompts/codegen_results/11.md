// internal/infrastructure/persistence/postgres/cms_repository.go
package postgres

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"

	"github.com/aimv/media-gallery/internal/domain/cms"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

type CMSRepository struct {
	pool *pgxpool.Pool
}

func NewCMSRepository(pool *pgxpool.Pool) *CMSRepository {
	return &CMSRepository{pool: pool}
}

// Save создаёт или обновляет контентный блок.
// Поле Title сериализуется внутрь JSONB-колонки content.
func (r *CMSRepository) Save(ctx context.Context, block *cms.ContentBlock) error {
	contentMap := make(map[string]any, len(block.Content)+1)
	for k, v := range block.Content {
		contentMap[k] = v
	}
	contentMap["title"] = block.Title

	contentJSON, err := json.Marshal(contentMap)
	if err != nil {
		return fmt.Errorf("marshal content: %w", err)
	}

	_, err = r.pool.Exec(ctx, `
		INSERT INTO content_blocks (
			id, block_type, content, media_id, created_at, updated_at
		) VALUES ($1, $2, $3, $4, now(), now())
		ON CONFLICT (id) DO UPDATE SET
			block_type = EXCLUDED.block_type,
			content = EXCLUDED.content,
			media_id = EXCLUDED.media_id,
			updated_at = now()
	`,
		block.ID,
		block.BlockType,
		contentJSON,
		block.MediaID,
	)
	if err != nil {
		return fmt.Errorf("upsert content block: %w", err)
	}
	return nil
}

// GetByIDWithMedia возвращает контентный блок с присоединёнными метаданными медиафайла.
// Поле Title извлекается из JSONB-колонки content.
func (r *CMSRepository) GetByIDWithMedia(ctx context.Context, id string) (*cms.BlockWithMedia, error) {
	var (
		result       cms.BlockWithMedia
		contentRaw   []byte
		mediaAssetID *string
		mediaOrig    *string
		mediaType    *string
		mediaStatus  *string
	)

	err := r.pool.QueryRow(ctx, `
		SELECT
			cb.id,
			cb.block_type,
			cb.content,
			cb.media_id,
			cb.created_at,
			cb.updated_at,
			ma.id,
			ma.original_file_name,
			ma.media_type,
			ma.status
		FROM content_blocks cb
		LEFT JOIN media_assets ma ON cb.media_id = ma.id
		WHERE cb.id = $1
	`, id).Scan(
		&result.ID,
		&result.BlockType,
		&contentRaw,
		&result.MediaID,
		&result.CreatedAt,
		&result.UpdatedAt,
		&mediaAssetID,
		&mediaOrig,
		&mediaType,
		&mediaStatus,
	)
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return nil, fmt.Errorf("content block not found: %w", err)
		}
		return nil, fmt.Errorf("select content block: %w", err)
	}

	var contentMap map[string]any
	if err := json.Unmarshal(contentRaw, &contentMap); err != nil {
		return nil, fmt.Errorf("unmarshal content: %w", err)
	}
	if title, ok := contentMap["title"].(string); ok {
		result.Title = title
		delete(contentMap, "title")
	}
	result.Content = contentMap

	if mediaAssetID != nil {
		result.Media = &cms.MediaInfo{
			ID:               *mediaAssetID,
			OriginalFileName: *mediaOrig,
			MediaType:        *mediaType,
			Status:           *mediaStatus,
		}
	}

	return &result, nil
}

// List возвращает все контентные блоки без медиа-джойна.
func (r *CMSRepository) List(ctx context.Context) ([]*cms.ContentBlock, error) {
	rows, err := r.pool.Query(ctx, `
		SELECT id, block_type, content, media_id, created_at, updated_at
		FROM content_blocks
		ORDER BY created_at DESC
	`)
	if err != nil {
		return nil, fmt.Errorf("list content blocks: %w", err)
	}
	defer rows.Close()

	var blocks []*cms.ContentBlock
	for rows.Next() {
		var (
			b          cms.ContentBlock
			contentRaw []byte
		)
		if err := rows.Scan(
			&b.ID,
			&b.BlockType,
			&contentRaw,
			&b.MediaID,
			&b.CreatedAt,
			&b.UpdatedAt,
		); err != nil {
			return nil, fmt.Errorf("scan content block: %w", err)
		}

		var contentMap map[string]any
		if err := json.Unmarshal(contentRaw, &contentMap); err != nil {
			return nil, fmt.Errorf("unmarshal content: %w", err)
		}
		if title, ok := contentMap["title"].(string); ok {
			b.Title = title
			delete(contentMap, "title")
		}
		b.Content = contentMap

		blocks = append(blocks, &b)
	}
	return blocks, rows.Err()
}

// Delete удаляет контентный блок по ID.
func (r *CMSRepository) Delete(ctx context.Context, id string) error {
	_, err := r.pool.Exec(ctx, `DELETE FROM content_blocks WHERE id = $1`, id)
	if err != nil {
		return fmt.Errorf("delete content block: %w", err)
	}
	return nil
}
